#include <iostream>

using namespace std;
enum BusType{CAN,LIN,UART};
class CommInterface{
    public:
    uint id;
    uint baud_rate;
    uint timeout;
    
    CommInterface(uint x,uint y,uint z)
    {
        id=x;
        baud_rate=y;
        timeout=z;
    }
    void printCommInterface()
    {
        cout<<"Comm ID is : "<<id<<endl;
        cout<<"Comm baud rate is :"<<baud_rate<<endl;
        cout<<"Comm timeout is :"<<timeout<<endl;
    }
};
CommInterface createCommInterface(BusType type){
    if(type==CAN){
        return CommInterface(1,500,100);
    }
    else if(type==LIN){
        return CommInterface(2,20,200);
    }
    else{
        return CommInterface(3,115,50);
    }
}
int main()
{
    CommInterface can=createCommInterface(CAN);
    can.printCommInterface();
    CommInterface lin = createCommInterface(LIN);
    lin.printCommInterface();
    CommInterface uart = createCommInterface(UART);
    uart.printCommInterface();
    return 0;
}
